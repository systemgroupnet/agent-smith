# Test-only dependencies. Deliberately NOT in reviewer/requirements.txt: that file
# is installed into the production image, and pytest has no business there.
#
# The suite also needs the reviewer's own dependencies (langchain, deepagents,
# pydantic, …) because it imports the code under test. CI installs both:
#   pip install -r reviewer/requirements.txt -r tests/requirements.txt
pytest>=8.0,<10.0

# The collector's engagement/digest modules import SQLAlchemy for `text()`, so the
# suite needs it to cover them. Deliberately *not* psycopg: the engagement tests
# exercise the pure logic (thresholds, ISO weeks, formatting) and never open a
# connection, so CI needs no PostgreSQL. The SQL itself is verified against a real
# database by hand — see tests/README.md.
SQLAlchemy>=2.0,<3.0

# The migration tests load collector/migrations/versions/*.py to check the revision
# chain and that a migration and storage.py agree on the columns. Every migration
# imports `alembic.op` at module level, so loading one needs alembic installed —
# but no database: nothing is upgraded. Same pin as collector/requirements.txt.
alembic>=1.13,<2.0
